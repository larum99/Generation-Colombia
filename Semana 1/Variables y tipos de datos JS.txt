/*

1. Tipos de Datos Primitivos
Number: Valores numéricos enteros y decimales (ej. 42 o 3.14).
BigInt: Enteros de longitud arbitraria, ideales para operaciones matemáticas gigantescas (ej. 12345678901234567890n).
Boolean: Representa estados lógicos: true (verdadero) o false (falso).
Undefined: Una variable declarada a la que aún no se le ha asignado un valor.
Null: Ausencia intencional de cualquier valor.
Symbol: Un valor único e inmutable utilizado frecuentemente como identificador de propiedades de objetos.


String: Cadenas de texto. Van entre comillas simples o dobles (ej. 'Hola').Pueden ser comillas simples 'Hola', comillas dobles "Hola", o comillas invertidas `Hola` (backticks). Las primeras dos son iguales. Los backticks permiten poner variables dentro sin tener que cortar el texto, usando ${variable}. Esto se llama interpolación.
*/

//Operadores de comparacion 

/*
 Mayor que:  > 
 Menor que: <
 Menor o igual que: <= 
 Mayor o igual que: >= 
 igualdad: ===
 diferente: !==
*/

console.log("5 > 3:", 5 > 3);     // true  (5 es mayor que 3)
console.log("5 < 3:", 5 < 3);     // false (5 no es menor que 3)
console.log("5 >= 5:", 5 >= 5);   // true  (5 es mayor o igual que 5)
console.log("5 === 5:", 5 === 5); // true  (valor y tipo son iguales)
console.log("5 === '5':", 5 === '5'); // false (número vs texto)

let esMiembroPremium = true;            // ¿Es miembro premium?
let totalCompra = 150000;               // Total de la compra en pesos

let descuentoPremium = esMiembroPremium;               // true (es miembro)

// OPERADORES LÓGICOS: && (Y), || (O), ! (NO)

console.log("true && true =", true && true);    // true (ambos true)
console.log("true && false =", true && false);  // false (uno es false)
console.log("true || false =", true || false);  // true (al menos uno true)
console.log("false || false =", false || false); // false (ambos false)
console.log("!true =", !true);                  // false (invierte)
console.log("!false =", !false);                // true (invierte)

/* 
Los datos también pueden organizarse en estructuras, como arreglos(Arrays) u objetos; a estas se les llama estructuras de datos.
 Los arreglos y los objetos se utilizan para agrupar y gestionar colecciones de datos en JavaScript.

 2. Tipos de Datos Complejos 
Array: Un tipo especial de objeto utilizado para almacenar listas ordenadas de elementos bajo una sola variable (ej. [1, 2, 3]).
*/

let listaDeCompras = ["pan", "leche","chocolate","huevos", "queso"];

console.log("Lista de compras", listaDeCompras);
console.log();
console.log("Primer elemento (indice 0): ", listaDeCompras[0]);
console.log("Segundo elemento (índice 1): ", listaDeCompras[1]);
console.log(`Tercer elemento: ${listaDeCompras[2]}`);

console.log("Cantidad de items:", listaDeCompras.length);  

let ultimoIndice = listaDeCompras.length - 1;                  
console.log("Último elemento:", listaDeCompras[ultimoIndice]);

//Métodos para modificar arreglos


// push() - Agrega un elemento al FINAL

listaDeCompras.push("café");
console.log("Después de push('café'):", listaDeCompras);
console.log("Nueva longitud:", listaDeCompras.length);  

// pop() - Elimina el ÚLTIMO elemento y lo devuelve

let itemEliminado = listaDeCompras.pop();
console.log("Eliminé:", itemEliminado);                 
console.log("Después de pop():", listaDeCompras);       

// unshift() - Agrega un elemento al PRINCIPIO
listaDeCompras.unshift("jugo");
console.log("Después de unshift('jugo'):", listaDeCompras);

// shift() - Elimina el PRIMER elemento
let primerItem = listaDeCompras.shift();
console.log("Eliminé el primero:", primerItem);         
console.log("Después de shift():", listaDeCompras);

/*
Objectos: Un object  es una estructura de datos que guarda información en pares de clave: valor (key: value). A diferencia de los arreglos que usan índices numéricos, los objetos usan nombres (claves) para acceder a los valores. Las claves son strings. Los valores pueden ser cualquier tipo de dato: number, string, boolean, array, otro objeto. Los objetos se definen con llaves {}. Para acceder a un valor, se puede usar objeto.clave (notación de punto) o objeto["clave"] (notación de corchetes)
*/

//Perfil de usuario

let usuario = {
  nombre: "Ana",      // clave: "nombre", valor: "Ana"
  edad: 25,           // clave: "edad", valor: 25
  ciudad: "Bogotá"    // clave: "ciudad", valor: "Bogotá"
};

console.log("Objeto completo:", usuario);

// Notación de punto 
console.log("Nombre (notación punto):", usuario.nombre);   // Ana
console.log("Edad (notación punto):", usuario.edad);       // 25

// Notación de corchetes
console.log("Ciudad (notación corchetes):", usuario["ciudad"]);  // Bogotá

// Modificar una propiedad existente
usuario.edad = 26;      // Cambio la edad de 25 a 26
console.log("Edad actualizada:", usuario.edad);

// Agregar una nueva propiedad
usuario.profesion = "Programadora";  // Creo la clave "profesion" con su valor
console.log("Después de agregar profesión:", usuario);

// Eliminar una propiedad 
delete usuario.ciudad;   // Borro la clave "ciudad" y su valor
console.log("Después de eliminar ciudad:", usuario);

//Ejemplo objeto 2

let producto = {
  nombre: "Laptop Gamer",                    // String: nombre del producto
  precio: 2500000,                           // Number: precio en pesos
  disponible: true,                          // Boolean: true = hay stock
  etiquetas: ["electrónica", "computadores", "oferta"],  // Array: lista de etiquetas
  especificaciones: {                        // Objeto anidado: características técnicas
    marca: "Lenovo",
    ram: "16GB",
    almacenamiento: "512GB SSD"
  }
};

console.log("=== PRODUCTO ===");
console.log("Nombre:", producto.nombre);                     // String
console.log("Precio: $" + producto.precio);                  // Number
console.log("Disponible:", producto.disponible);             // Boolean
console.log("Primera etiqueta:", producto.etiquetas[0]);     // Accedo al array dentro del objeto
console.log("Marca:", producto.especificaciones.marca);      // Accedo al objeto dentro del objeto
console.log("RAM:", producto.especificaciones.ram);          // Otro valor del objeto anidado

RESUMEN DE TIPOS DE DATOS:

NUMBER (número)
  → let precio = 1500;
  → Operaciones: + - * / % **

 STRING (texto)
  → let nombre = "Ana";
  → Métodos: .toLowerCase(), .toUpperCase(), .trim(), .length

  
BOOLEAN (verdadero/falso)
  → let puedeEntrar = edad >= 18;
  → Operadores: && (Y), || (O), ! (NO)

ARRAY (lista ordenada)
  → let lista = ["pan", "leche"];
  → Acceso: lista[0] (empieza en 0)
  → Métodos: .push(), .pop(), .length

OBJECT (pares clave-valor)
  → let usuario = {nombre: "Ana", edad: 25};
  → Acceso: usuario.nombre
*/
